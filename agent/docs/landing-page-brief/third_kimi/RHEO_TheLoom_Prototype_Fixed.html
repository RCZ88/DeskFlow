(function(){ const svgNS = 'http://www.w3.org/2000/svg'; const W = 1400, H = 800; const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches; // Seven warp threads = seven subsystems. "over" alternates with "under" so the // weft reads as an actual weave, not just a line crossing other lines. const warps = [ { x:100, label:'TIME', over:true }, { x:300, label:'MONEY', over:false }, { x:500, label:'FOCUS', over:true }, { x:700, label:'LEARNING', over:false }, { x:900, label:'CHAT', over:true }, { x:1100, label:'TERMINAL', over:false }, { x:1300, label:'TIMELINE', over:true }, ]; const warpGroup = document.getElementById('warpGroup'); warps.forEach(w=>{ const line = document.createElementNS(svgNS,'line'); line.setAttribute('x1', w.x); line.setAttribute('x2', w.x); line.setAttribute('y1', 40); line.setAttribute('y2', 720); line.classList.add('warp'); line.dataset.label = w.label; warpGroup.appendChild(line); const label = document.createElementNS(svgNS,'text'); label.setAttribute('x', w.x); label.setAttribute('y', 752); label.classList.add('warp-label','mono'); label.textContent = w.label; warpGroup.appendChild(label); }); // The weft (AI thread) is one gentle wave across the whole width, not a rigid // straight line — closer to how an actual thread sits under slight tension. function wavePath(y0, amp, freq, steps){ let d = `M 0 ${y0.toFixed(1)}`; for(let i=1;i<=steps;i++){ const x = (W/steps)*i; const y = y0 + amp*Math.sin((x/W)*Math.PI*freq); d += ` L ${x.toFixed(1)} ${y.toFixed(1)}`; } return d; } const weftY = 380, amp = 26, freq = 2.6; const dAttr = wavePath(weftY, amp, freq, 140); const weftBase = document.getElementById('weftBase'); const weftTop = document.getElementById('weftTop'); weftBase.setAttribute('d', dAttr); weftTop.setAttribute('d', dAttr); const totalLen = weftBase.getTotalLength(); [weftBase, weftTop].forEach(p=>{ p.style.strokeDasharray = totalLen; p.style.strokeDashoffset = totalLen; // fully hidden until revealed }); // --- the actual "weave" trick --- // weftBase and weftTop carry the IDENTICAL path. Which one is visible at a // given x is decided by two complementary clip-paths built from the warps' // over/under flags — that's what makes the same thread appear to duck behind // one warp and pop in front of the next, using only two static clip regions // instead of per-frame z-index juggling. const bandHalf = 24; const overRects = warps.filter(w=>w.over).map(w=>({ x:w.x-bandHalf, w:bandHalf*2 })); const sorted = overRects.slice().sort((a,b)=>a.x-b.x); let cursor = 0; const underRects = []; sorted.forEach(r=>{ if(r.x > cursor) underRects.push({ x:cursor, w:r.x-cursor }); cursor = r.x + r.w; }); if(cursor < W) underRects.push({ x:cursor, w:W-cursor }); function fillClip(id, rects){ const clip = document.getElementById(id); rects.forEach(r=>{ const rect = document.createElementNS(svgNS,'rect'); rect.setAttribute('x', r.x); rect.setAttribute('y', 0); rect.setAttribute('width', r.w); rect.setAttribute('height', H); clip.appendChild(rect); }); } fillClip('clipOver', overRects); fillClip('clipUnder', underRects); // small glow dots at every crossing, fading in as the weft's reveal passes them const dotGroup = document.getElementById('dotGroup'); const dots = warps.map(w=>{ const y = weftY + amp*Math.sin((w.x/W)*Math.PI*freq); const c = document.createElementNS(svgNS,'circle'); c.setAttribute('cx', w.x); c.setAttribute('cy', y); c.setAttribute('r', 5); c.classList.add('dot'); dotGroup.appendChild(c); return { el:c, progress: w.x / W }; }); // caption boxes sit roughly under their matching warp thread const capMap = { MONEY:'cap1', LEARNING:'cap2', TERMINAL:'cap3' }; function positionCaptions(){ warps.forEach(w=>{ const id = capMap[w.label]; if(!id) return; const el = document.getElementById(id); const pct = (w.x / W) * 100; el.style.left = `calc(${pct}% - 70px)`; el.style.top = '64%'; }); } positionCaptions(); window.addEventListener('resize', positionCaptions); // Note: precise alignment assumes a roughly widescreen viewport, since the SVG // uses preserveAspectRatio="slice" for full-bleed cover. Good enough for a // desktop prototype; a production build would tie caption position to the // SVG's actual rendered warp coordinates via getBoundingClientRect(). // ---- reduced motion: skip all animation, show the finished state directly ---- if(reduceMotion){ document.getElementById('reducedNote').style.display = 'block'; [weftBase, weftTop].forEach(p=> p.style.strokeDashoffset = 0); dots.forEach(d=> d.el.style.opacity = 0.9); const hero = document.getElementById('heroCopy'); hero.style.opacity = 1; hero.style.transform = 'none'; ['cap1','cap2','cap3'].forEach(id=> document.getElementById(id).style.opacity = 0.95); return; } gsap.registerPlugin(ScrollTrigger); // hero text fades in on load, independent of scroll gsap.to('#heroCopy', { opacity:1, y:0, duration:1, delay:0.3, ease:'power2.out' }); function applyReveal(v){ const offset = totalLen * (1 - v); weftBase.style.strokeDashoffset = offset; weftTop.style.strokeDashoffset = offset; dots.forEach(d=>{ d.el.style.opacity = gsap.utils.clamp(0, 1, (v - d.progress + 0.05) * 12); }); } const state = { reveal: 0 }; const tl = gsap.timeline({ scrollTrigger: { trigger: '.scroll-space', start: 'top top', end: 'bottom bottom', scrub: 1, pin: '#stage', } }); // beat 1: the weave draws itself in as you scroll tl.to(state, { reveal:1, duration:0.34, onUpdate:()=>applyReveal(state.reveal) }); tl.to('#heroCopy', { opacity:0, y:-16, duration:0.06 }, '<90%'); tl.to('#warpGroup', { opacity:0.4, duration:0.08 }); // beats 2–4: the shuttle "visits" three threads in turn, each with its caption const beats = [ { warp:'MONEY', cap:'cap1' }, { warp:'LEARNING', cap:'cap2' }, { warp:'TERMINAL', cap:'cap3' }, ]; beats.forEach(b=>{ const lineEl = [...warpGroup.querySelectorAll('.warp')].find(l => l.dataset.label === b.warp); tl.to(lineEl, { attr:{ stroke:'#fbbf24' }, opacity:1, strokeWidth:3, duration:0.14 }, '>') .to(`#${b.cap}`, { opacity:1, duration:0.1 }, '<') .to(lineEl, { opacity:0.4, strokeWidth:1.5, duration:0.1 }, '+=0.06') .to(`#${b.cap}`, { opacity:0, duration:0.08 }, '<'); }); tl.to('#warpGroup', { opacity:0.9, duration:0.1 }); })();
